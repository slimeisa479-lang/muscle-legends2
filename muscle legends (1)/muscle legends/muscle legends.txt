-- WindUI bootstrap. Uses the published WindUI distribution.
-- Try game:HttpGet first, then the executor request interface as fallback.
local WINDUI_URL = "https://raw.githubusercontent.com/Footagesus/WindUI/main/dist/main.lua"
local function fetchWindUISource()
    local errors = {}

    local ok, result = pcall(function()
        return game:HttpGet(WINDUI_URL)
    end)
    if ok and type(result) == "string" and #result > 100 then
        return result
    end
    table.insert(errors, "game:HttpGet: " .. tostring(result))

    local requestFunction = request or http_request
    if type(requestFunction) == "function" then
        local requestOK, response = pcall(function()
            return requestFunction({Url = WINDUI_URL, Method = "GET"})
        end)
        if requestOK and type(response) == "table" then
            local body = response.Body or response.body
            if type(body) == "string" and #body > 100 then
                return body
            end
            table.insert(errors, "request returned no response body")
        else
            table.insert(errors, "request: " .. tostring(response))
        end
    end

    error("Silence could not download WindUI. Check Delta X HTTP/loadstring support and internet access. " .. table.concat(errors, " | "), 0)
end

local compile = loadstring
if type(compile) ~= "function" then
    error("Silence cannot start: this Delta X build does not expose loadstring.", 0)
end

local source = fetchWindUISource()
local compiled, compileError = compile(source)
if type(compiled) ~= "function" then
    error("Silence WindUI compile error: " .. tostring(compileError), 0)
end

local loaded, WindUI = pcall(compiled)
if not loaded or type(WindUI) ~= "table" or type(WindUI.CreateWindow) ~= "function" then
    error("Silence WindUI initialization failed: " .. tostring(WindUI), 0)
end

local runtime = {Alive = true, Threads = {}, Connections = {}, Cleanups = {}}
shared.SilenceMuscleLegendsRuntime = runtime

function runtime:Spawn(callback, ...)
    if not self.Alive then return end
    local thread
    thread = coroutine.create(function(...)
        local success, result = pcall(callback, ...)
        self.Threads[thread] = nil
        if not success then self.LastError = result end
    end)
    self.Threads[thread] = true
    task.spawn(thread, ...)
    return thread
end

function runtime:Connect(signal, callback)
    local connection = signal:Connect(function(...) if self.Alive then callback(...) end end)
    table.insert(self.Connections, connection)
    return connection
end

function runtime:Defer(callback, ...)
    if not self.Alive then return end
    return task.defer(callback, ...)
end

function runtime:Delay(delayTime, callback, ...)
    if not self.Alive then return end
    return task.delay(delayTime, callback, ...)
end

function runtime:AddCleanup(callback)
    table.insert(self.Cleanups, callback)
end

function runtime:Destroy()
    if not self.Alive then return end
    self.Alive = false
    for _, connection in ipairs(self.Connections) do pcall(function() connection:Disconnect() end) end
    for index = #self.Cleanups, 1, -1 do pcall(self.Cleanups[index]) end
    for thread in pairs(self.Threads) do if thread ~= coroutine.running() then pcall(task.cancel, thread) end end
    table.clear(self.Connections)
    table.clear(self.Threads)
    if shared.SilenceMuscleLegendsRuntime == self then shared.SilenceMuscleLegendsRuntime = nil end
end
