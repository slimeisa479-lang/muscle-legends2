-- WindUI bootstrap (pinned stable release; avoid the moving main branch).
-- Delta X Android may expose either game:HttpGet or an executor request API.
local WINDUI_URLS = {
    "https://github.com/Footagesus/WindUI/releases/download/1.6.66/main.lua",
    "https://raw.githubusercontent.com/Footagesus/WindUI/main/dist/main.lua",
}

local function getRequestFunction()
    if type(request) == "function" then return request end
    if type(http_request) == "function" then return http_request end
    if type(syn) == "table" and type(syn.request) == "function" then return syn.request end
    if type(http) == "table" and type(http.request) == "function" then return http.request end
    return nil
end

local function fetchWindUISource()
    local errors = {}
    local requestFunction = getRequestFunction()

    for _, url in ipairs(WINDUI_URLS) do
        if requestFunction then
            local ok, response = pcall(requestFunction, {Url = url, Method = "GET"})
            if ok and type(response) == "table" then
                local body = response.Body or response.body or response.ResponseBody
                local status = tonumber(response.StatusCode or response.Status or response.status_code)
                if type(body) == "string" and #body > 100 and (not status or (status >= 200 and status < 300)) then
                    return body
                end
                table.insert(errors, "request response invalid for " .. url .. " (status=" .. tostring(status) .. ")")
            else
                table.insert(errors, "request failed for " .. url .. ": " .. tostring(response))
            end
        end

        local ok, body = pcall(function()
            return game:HttpGet(url)
        end)
        if ok and type(body) == "string" and #body > 100 then
            return body
        end
        table.insert(errors, "HttpGet failed for " .. url .. ": " .. tostring(body))
    end

    error("Silence: WindUI download failed. Delta X may be blocking GitHub or HTTP requests. " .. table.concat(errors, " | "), 0)
end

local compile = loadstring
if type(compile) ~= "function" then
    error("Silence: loadstring is unavailable in this Delta X session; the menu cannot be created.", 0)
end

local source = fetchWindUISource()
local compiled, compileError = compile(source)
if type(compiled) ~= "function" then
    error("Silence: WindUI source could not compile: " .. tostring(compileError), 0)
end

local loaded, WindUI = pcall(compiled)
if not loaded then
    error("Silence: WindUI failed during initialization: " .. tostring(WindUI), 0)
end
if type(WindUI) ~= "table" or type(WindUI.CreateWindow) ~= "function" then
    error("Silence: downloaded WindUI did not expose CreateWindow.", 0)
end

local library = WindUI

local function enableAntiAFK()
    local player = game:GetService("Players").LocalPlayer
    local virtualUser = game:GetService("VirtualUser")
    player.Idled:Connect(function()
        virtualUser:CaptureController()
        virtualUser:ClickButton2(Vector2.new())
    end)
end

library.EnableAntiAFK = function()
    enableAntiAFK()
end

local runtime = {Alive = true, Threads = {}, Connections = {}, Cleanups = {}}
shared.SilenceMuscleLegendsRuntime = runtime

function runtime:Spawn(callback, ...)
    if not self.Alive then return end
    local thread
    thread = coroutine.create(function(...)
        local success, result = pcall(callback, ...)
        self.Threads[thread] = nil
        if not success then self.LastError = result end
    end)
    self.Threads[thread] = true
    task.spawn(thread, ...)
    return thread
end

function runtime:Connect(signal, callback)
    local connection = signal:Connect(function(...) if self.Alive then callback(...) end end)
    table.insert(self.Connections, connection)
    return connection
end

function runtime:Defer(callback, ...)
    if not self.Alive then return end
    return task.defer(callback, ...)
end

function runtime:Delay(delayTime, callback, ...)
    if not self.Alive then return end
    return task.delay(delayTime, callback, ...)
end
